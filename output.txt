<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\UserAccessControlModel;
use App\Services\PhotoStudio\PhotoStudioService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Session;
use Illuminate\Support\Facades\Validator;
use Symfony\Component\HttpFoundation\BinaryFileResponse;
use Symfony\Component\HttpFoundation\Response;
use Symfony\Component\HttpFoundation\UploadedFile;
use Throwable;

class PhotoStudioController extends Controller
{
    private const MODULE_ID = 29;
    private const SESSION_PREFIX = 'photo_studio.pending.';
    private const PREVIEW_LIFETIME = 1800;

    private function allowedTypes(): array
    {
        return ['staff', 'office_chief', 'home_officer'];
    }

    private function authorizePhotoStudio(string $type)
    {
        $user = Auth::user();

        if (!$user) {
            abort(401);
        }

        if (!UserAccessControlModel::getAccessAddByModuleAndUserMapId(
            ($type === 'home_officer' ? 1 : self::MODULE_ID),
            $user->id
        )) {
            abort(403, 'You do not have permission to use Photo Studio.');
        }

        return $user;
    }

    private function sessionKey(string $type): string
    {
        abort_unless(in_array($type, $this->allowedTypes(), true), 404);

        return self::SESSION_PREFIX . $type;
    }

    private function pendingPreview(string $type): ?array
    {
        $pending = Session::get($this->sessionKey($type));

        if (!is_array($pending)) {
            return null;
        }

        $user = Auth::user();

        if (
            !$user ||
            !isset($pending['token'], $pending['user_id'], $pending['created_at']) ||
            !is_string($pending['token']) ||
            !preg_match('/^[a-f0-9]{48}$/', $pending['token']) ||
            (int) $pending['user_id'] !== (int) $user->id ||
            !is_numeric($pending['created_at']) ||
            (time() - (int) $pending['created_at']) > self::PREVIEW_LIFETIME ||
            (int) $pending['created_at'] > time()
        ) {
            return null;
        }

        return $pending;
    }

    /**
     * Generate a private AI preview for Staff or Office Chief.
     */
    public function createPreview(Request $request, string $type, PhotoStudioService $studio)
    {
        $user = $this->authorizePhotoStudio($type);
        $this->sessionKey($type);

        $validator = Validator::make($request->all(), [
            'attachment' => 'required|file|mimes:jpg,jpeg,png|max:1024',
        ]);

        if ($validator->fails()) {
            return response()->json([
                'status' => 'error',
                'message' => $validator->errors()->first('attachment'),
            ], 422);
        }

        $oldPending = $this->pendingPreview($type);
        $newToken = null;

        try {
            $newToken = $studio->createPreview(
                $request->file('attachment')
            );

            $studio->getPreviewPath($newToken);

            Session::put($this->sessionKey($type), [
                'token' => $newToken,
                'user_id' => (int) $user->id,
                'created_at' => time(),
            ]);

            if ($oldPending && $oldPending['token'] !== $newToken) {
                $studio->delete($oldPending['token']);
            }

            return response()->json([
                'status' => 'success',
                'token' => $newToken,
                'preview_url' => route('admin.photo-studio.preview', [
                    'type' => $type,
                    'token' => $newToken,
                ]),
                'expires_in' => self::PREVIEW_LIFETIME,
            ]);
        } catch (Throwable $exception) {
            if ($newToken) {
                $studio->delete($newToken);
            }

            report($exception);

            return response()->json([
                'status' => 'error',
                'message' => 'Photo processing failed. Please try another photo.',
            ], 422);
        }
    }

    /**
     * Display a preview only to the Admin session that created it.
     */
    public function showPreview(
        string $type,
        string $token,
        PhotoStudioService $studio
    ): BinaryFileResponse {
        $this->authorizePhotoStudio($type);
        $this->sessionKey($type);

        $pending = $this->pendingPreview($type);

        abort_unless(
            $pending && hash_equals($pending['token'], $token),
            404
        );

        try {
            $path = $studio->getPreviewPath($token);
        } catch (Throwable $exception) {
            abort(404);
        }

        return response()->file($path, [
            'Content-Type' => 'image/jpeg',
            'Cache-Control' => 'private, no-store, no-cache, must-revalidate, max-age=0',
            'Pragma' => 'no-cache',
            'X-Content-Type-Options' => 'nosniff',
        ]);
    }

    /**
     * Reject and remove a pending private preview.
     */
    public function rejectPreview(
        Request $request,
        string $type,
        PhotoStudioService $studio
    ): Response {
        $this->authorizePhotoStudio($type);
        $key = $this->sessionKey($type);

        $validator = Validator::make($request->all(), [
            'token' => 'required|string|regex:/^[a-f0-9]{48}$/',
        ]);

        if ($validator->fails()) {
            return response()->json([
                'status' => 'error',
                'message' => 'Invalid preview reference.',
            ], 422);
        }

        $pending = $this->pendingPreview($type);
        $token = (string) $request->input('token');

        abort_unless(
            $pending && hash_equals($pending['token'], $token),
            404
        );

        $studio->delete($token);
        Session::forget($key);

        return response()->json([
            'status' => 'success',
            'message' => 'Photo preview rejected.',
        ]);
    }
}

<?php

namespace App\Services\PhotoStudio;

use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\Session;
use RuntimeException;
use Symfony\Component\Process\Process;

class PhotoStudioService
{
    private string $workspace;
    private string $python;
    private string $processor;

    public function __construct()
    {
        $this->workspace = rtrim(sys_get_temp_dir(), '/')
            . '/dor-photo-studio';

        $this->python = (string) config('photo-studio.python');

        if ($this->python === '' || !is_executable($this->python)) {
            throw new RuntimeException('Photo Studio Python executable is not configured or executable.');
        }
        $this->processor = base_path(
            'app/Services/PhotoStudio/process_photo.py'
        );

        if (!is_dir($this->workspace)) {
            if (!mkdir($this->workspace, 0700, true) && !is_dir($this->workspace)) {
                throw new RuntimeException('Private AI workspace could not be created.');
            }
        }

        chmod($this->workspace, 0700);
    }

    /**
     * Save an uploaded original privately and generate its AI preview.
     * The original upload is never modified.
     */
    public function createPreview(UploadedFile $upload): string
    {
        if (!$upload->isValid()) {
            throw new RuntimeException('The uploaded photo is invalid.');
        }

        if ($upload->getSize() > 1024 * 1024) {
            throw new RuntimeException('Photo size must not exceed 1 MB.');
        }

        $mime = (new \finfo(FILEINFO_MIME_TYPE))
            ->file($upload->getRealPath());

        $extension = match ($mime) {
            'image/jpeg' => 'jpg',
            'image/png' => 'png',
            default => throw new RuntimeException('Only JPEG and PNG photos are supported.'),
        };

        $imageInfo = getimagesize($upload->getRealPath());

        if (
            !$imageInfo ||
            $imageInfo[0] < 100 ||
            $imageInfo[1] < 100 ||
            ($imageInfo[0] * $imageInfo[1]) > 25000000
        ) {
            throw new RuntimeException('Photo dimensions are invalid or too large.');
        }

        $id = bin2hex(random_bytes(24));
        $directory = $this->workspace . '/' . $id;

        if (!mkdir($directory, 0700)) {
            throw new RuntimeException('Private photo workspace could not be created.');
        }

        chmod($directory, 0700);

        $originalPath = $directory . '/original.' . $extension;
        $previewPath = $directory . '/preview.jpg';

        try {
            if (!copy($upload->getRealPath(), $originalPath)) {
                throw new RuntimeException('Original photo could not be stored privately.');
            }

            chmod($originalPath, 0600);

            $process = new Process([
                $this->python,
                $this->processor,
                '--input',
                $originalPath,
                '--output',
                $previewPath,
            ]);

            $process->setTimeout(180);
            $process->run();

            if (!$process->isSuccessful() || !is_file($previewPath)) {
                throw new RuntimeException('AI photo processing failed. Please try again.');
            }

            chmod($previewPath, 0600);

            return $id;
        } catch (\Throwable $exception) {
            $this->delete($id);
            throw $exception;
        }
    }

    /**
     * Return a private preview path after validating its token.
     */
    public function getPreviewPath(string $id): string
    {
        $directory = $this->getDirectory($id);
        $path = $directory . '/preview.jpg';

        if (!is_file($path)) {
            throw new RuntimeException('AI preview is no longer available.');
        }

        return $path;
    }

    /**
     * Return the preserved original upload path.
     */
    public function getOriginalPath(string $id): string
    {
        $directory = $this->getDirectory($id);

        foreach (['jpg', 'png'] as $extension) {
            $path = $directory . '/original.' . $extension;

            if (is_file($path)) {
                return $path;
            }
        }

        throw new RuntimeException('Original photo is no longer available.');
    }

    /**
     * Remove one private preview workspace.
     */
    /**
     * Return an accepted AI preview as a temporary uploaded file.
     * The preview must belong to the current Admin session and user.
     */
    public function getAcceptedUpload(
        string $type,
        string $token,
        int $userId
    ): UploadedFile {
        $this->assertPendingPreview($type, $token, $userId);

        $path = $this->getPreviewPath($token);

        return new UploadedFile(
            $path,
            'photo-studio-preview.jpg',
            'image/jpeg',
            UPLOAD_ERR_OK,
            true
        );
    }

    /**
     * Delete the private original and preview after successful database save.
     */
    public function completeAcceptedPreview(
        string $type,
        string $token,
        int $userId
    ): void {
        $this->assertPendingPreview($type, $token, $userId);

        $this->delete($token);
        Session::forget('photo_studio.pending.' . $type);
    }

    private function assertPendingPreview(
        string $type,
        string $token,
        int $userId
    ): void {
        if (!in_array($type, ['staff', 'office_chief', 'home_officer'], true)) {
            throw new RuntimeException('Invalid Photo Studio type.');
        }

        if (!preg_match('/^[a-f0-9]{48}$/', $token)) {
            throw new RuntimeException('Invalid photo preview reference.');
        }

        $pending = Session::get('photo_studio.pending.' . $type);

        if (
            !is_array($pending) ||
            !isset($pending['token'], $pending['user_id'], $pending['created_at']) ||
            !is_string($pending['token']) ||
            !hash_equals($pending['token'], $token) ||
            (int) $pending['user_id'] !== $userId ||
            !is_numeric($pending['created_at']) ||
            (time() - (int) $pending['created_at']) > 1800 ||
            (int) $pending['created_at'] > time()
        ) {
            throw new RuntimeException('Photo preview is expired or does not belong to this Admin session.');
        }
    }
    public function delete(string $id): void
    {
        if (!preg_match('/^[a-f0-9]{48}$/', $id)) {
            return;
        }

        $directory = $this->workspace . '/' . $id;

        if (!is_dir($directory)) {
            return;
        }

        foreach (glob($directory . '/*') ?: [] as $file) {
            if (is_file($file)) {
                @unlink($file);
            }
        }

        @rmdir($directory);
    }

    private function getDirectory(string $id): string
    {
        if (!preg_match('/^[a-f0-9]{48}$/', $id)) {
            throw new RuntimeException('Invalid photo preview reference.');
        }

        return $this->workspace . '/' . $id;
    }
}

#!/usr/bin/env python3

import argparse
import json
import math
import os
import sys
from io import BytesIO
from pathlib import Path

import cv2
import numpy as np
from PIL import Image, ImageOps
from rembg import new_session, remove


FACE_MODEL = Path.home() / ".u2net" / "face_detection_yunet_2026may.onnx"
MAX_SIDE = 2200


def straighten(image):
    """Use face eye landmarks for gentle, non-cropping rotation."""
    if not FACE_MODEL.is_file():
        return image, 0.0, False

    try:
        h, w = image.shape[:2]
        detector = cv2.FaceDetectorYN.create(
            str(FACE_MODEL), "", (w, h),
            0.6, 0.3, 5000
        )
        detector.setInputSize((w, h))
        _, faces = detector.detect(image)

        if faces is None or len(faces) == 0:
            return image, 0.0, False

        # Use the largest detected face.
        face = max(faces, key=lambda f: float(f[2] * f[3]))

        # YuNet eye landmarks: indices 4-7.
        eye1 = (float(face[4]), float(face[5]))
        eye2 = (float(face[6]), float(face[7]))

        if abs(eye2[0] - eye1[0]) < 1:
            return image, 0.0, True

        left, right = sorted((eye1, eye2), key=lambda p: p[0])
        angle = math.degrees(math.atan2(
            right[1] - left[1],
            right[0] - left[0]
        ))

        if not (1.5 <= abs(angle) <= 12):
            return image, 0.0, True

        # Rotate the complete image on an expanded canvas.
        center = (w / 2, h / 2)
        matrix = cv2.getRotationMatrix2D(center, angle, 1.0)

        cos_a = abs(matrix[0, 0])
        sin_a = abs(matrix[0, 1])
        new_w = int(h * sin_a + w * cos_a)
        new_h = int(h * cos_a + w * sin_a)

        matrix[0, 2] += new_w / 2 - center[0]
        matrix[1, 2] += new_h / 2 - center[1]

        rotated = cv2.warpAffine(
            image, matrix, (new_w, new_h),
            flags=cv2.INTER_CUBIC,
            borderMode=cv2.BORDER_REFLECT
        )
        return rotated, round(angle, 2), True

    except Exception:
        # Face detection must never prevent photo processing.
        return image, 0.0, False


def enhance(image):
    """Conservative photographic enhancement."""
    # Reduce noise gently.
    denoised = cv2.fastNlMeansDenoisingColored(
        image, None, 3, 3, 7, 21
    )

    # Improve local contrast without aggressive colour changes.
    lab = cv2.cvtColor(denoised, cv2.COLOR_BGR2LAB)
    light, a, b = cv2.split(lab)

    clahe = cv2.createCLAHE(
        clipLimit=1.3,
        tileGridSize=(8, 8)
    )
    light = clahe.apply(light)

    enhanced = cv2.cvtColor(
        cv2.merge((light, a, b)),
        cv2.COLOR_LAB2BGR
    )

    # Gently compress extreme neutral highlights.
    hsv = cv2.cvtColor(enhanced, cv2.COLOR_BGR2HSV)
    h, s, v = cv2.split(hsv)

    highlight_mask = (v > 232) & (s < 45)
    v_float = v.astype(np.float32)
    v_float[highlight_mask] = (
        232 + (v_float[highlight_mask] - 232) * 0.45
    )
    v = np.clip(v_float, 0, 255).astype(np.uint8)

    enhanced = cv2.cvtColor(
        cv2.merge((h, s, v)),
        cv2.COLOR_HSV2BGR
    )

    # Mild sharpening; avoid aggressive edge enhancement.
    blurred = cv2.GaussianBlur(enhanced, (0, 0), 1.0)
    sharpened = cv2.addWeighted(
        enhanced, 1.35, blurred, -0.35, 0
    )

    return np.clip(sharpened, 0, 255).astype(np.uint8)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", required=True)
    parser.add_argument("--output", required=True)
    args = parser.parse_args()

    source = Path(args.input)
    destination = Path(args.output)

    if not source.is_file():
        raise ValueError("Input photo does not exist.")

    if not destination.parent.is_dir():
        raise ValueError("Output directory does not exist.")

    if source.stat().st_size > 10 * 1024 * 1024:
        raise ValueError("Input photo exceeds the processing limit.")

    with Image.open(source) as original:
        if original.format not in ("JPEG", "PNG"):
            raise ValueError("Only JPEG and PNG images are supported.")

        original = ImageOps.exif_transpose(original).convert("RGB")

        if original.width < 100 or original.height < 100:
            raise ValueError("Image dimensions are too small.")

        # Limit working dimensions for predictable memory use.
        original.thumbnail(
            (MAX_SIDE, MAX_SIDE),
            Image.Resampling.LANCZOS
        )

        rgb = np.array(original)

    bgr = cv2.cvtColor(rgb, cv2.COLOR_RGB2BGR)

    # Face-based gentle straightening, then image enhancement.
    bgr, rotation, face_detected = straighten(bgr)
    bgr = enhance(bgr)

    processed_rgb = cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB)

    prepared = BytesIO()
    Image.fromarray(processed_rgb).save(prepared, format="PNG")
    input_bytes = prepared.getvalue()

    # Safe fallback: do not remove background when no face is detected.
    # This protects templates, landscapes and non-person images.
    segmentation_used = False
    fallback_reason = "face_not_detected"

    if face_detected:
        try:
            session = new_session("u2net_human_seg")
            result_bytes = remove(
                input_bytes,
                session=session,
                post_process_mask=True
            )

            with Image.open(BytesIO(result_bytes)) as result:
                candidate = result.convert("RGBA")

            alpha = np.asarray(candidate.getchannel("A"))
            foreground_ratio = float(np.mean(alpha > 128))

            # Reject empty or nearly full-frame segmentation masks.
            if 0.02 <= foreground_ratio <= 0.98:
                foreground = candidate
                segmentation_used = True
                fallback_reason = None
            else:
                foreground = Image.fromarray(processed_rgb).convert("RGBA")
                fallback_reason = "implausible_segmentation_mask"

        except Exception:
            foreground = Image.fromarray(processed_rgb).convert("RGBA")
            fallback_reason = "segmentation_failed"
    else:
        foreground = Image.fromarray(processed_rgb).convert("RGBA")

    # Preserve the full photo and its proportions.
    foreground.thumbnail(
        (800, 800),
        Image.Resampling.LANCZOS
    )

    background = Image.new(
        "RGBA", (800, 800), (250, 250, 250, 255)
    )

    x = (800 - foreground.width) // 2
    y = (800 - foreground.height) // 2
    background.alpha_composite(foreground, (x, y))

    final_image = background.convert("RGB")

    temporary = destination.with_name(destination.name + ".tmp")
    final_image.save(
        temporary,
        format="JPEG",
        quality=92,
        optimize=True
    )
    os.replace(temporary, destination)

    print(json.dumps({
        "status": "success",
        "width": 800,
        "height": 800,
        "background": "#FAFAFA",
        "model": "u2net_human_seg",
        "face_detected": face_detected,
        "segmentation_used": segmentation_used,
        "fallback_reason": fallback_reason,
        "rotation_degrees": rotation,
        "enhancements": [
            "gentle_denoise",
            "local_contrast",
            "highlight_compression",
            "mild_sharpening"
        ],
        "output_bytes": destination.stat().st_size
    }))


if __name__ == "__main__":
    try:
        main()
    except Exception as exc:
        print(json.dumps({
            "status": "error",
            "message": str(exc)
        }), file=sys.stderr)
        sys.exit(1)
