<?php

namespace App\Http\Controllers;

use App\Models\ChatbotSetting;
use App\Models\Office;
use App\Services\Chatbot\ChatbotKnowledgeRetriever;
use App\Services\Chatbot\ChatbotProviderRouter;
use Illuminate\Http\Request;
use Throwable;

class PublicChatbotController extends Controller
{
    public function message(
        Request $request,
        string $subdomain,
        ChatbotKnowledgeRetriever $retriever,
        ChatbotProviderRouter $router
    ) {
        $settings = ChatbotSetting::current();

        abort_unless($settings->enabled, 404);

        $validated = $request->validate([
            'question' => ['required', 'string', 'min:2', 'max:2000'],
        ]);

        $officeId = (int) Office::getIdfromSubdomain($subdomain);

        abort_unless($officeId > 0, 404);

        $question = trim($validated['question']);

        try {
            $result = $retriever->search($question, $officeId, 5);
        } catch (Throwable $e) {
            report($e);

            return response()->json([
                'ok' => false,
                'message' => 'अहिले आधिकारिक स्रोत खोज्न सकिएन। कृपया पछि प्रयास गर्नुहोस्।',
            ], 503);
        }

        $matches = $result['matches'] ?? [];

        if (empty($matches)) {
            return response()->json([
                'ok' => true,
                'needs_consent' => true,
                'message' => 'DoR को आधिकारिक स्रोतमा जानकारी भेटिएन। बाह्य स्रोतबाट खोजौँ?',
                'sources' => [],
            ]);
        }

        $context = [];

        foreach (array_slice($matches, 0, 5) as $index => $match) {
            $context[] = "SOURCE " . ($index + 1)
                . "\nTitle: " . ($match['title'] ?? 'DoR document')
                . "\nContent: " . mb_substr(
                    (string) ($match['content'] ?? ''),
                    0,
                    2500
                );
        }

        $languageInstruction = match ($settings->response_language) {
            'ne' => 'Answer in Nepali.',
            'en' => 'Answer in English.',
            default => 'Answer in the language used by the question.',
        };

        $messages = [
            [
                'role' => 'system',
                'content' => 'You are the Department of Roads official information assistant. '
                    . 'Answer only from the supplied DoR official source excerpts. '
                    . 'Do not invent facts, dates, figures, policies or references. '
                    . 'If the excerpts do not answer the question, say so clearly. '
                    . $languageInstruction,
            ],
            [
                'role' => 'user',
                'content' => "Question:\n{$question}\n\n"
                    . "DoR Official Sources:\n"
                    . implode("\n\n", $context),
            ],
        ];

        $answer = null;
        $providerLabel = null;

        try {
            // Official-source answers must never fail over to external AI.
            $providerResult = $router->chat($messages, ['internal']);

            $answer = trim((string) ($providerResult['content'] ?? ''));

            if ($settings->provider_info_visible) {
                $providerLabel = $providerResult['provider_name']
                    ?? $providerResult['provider_code']
                    ?? null;
            }
        } catch (Throwable $e) {
            // If internal AI is unavailable, show retrieved official
            // excerpts instead of sending them to an external provider.
            report($e);
        }

        if (! filled($answer)) {
            $answer = implode("\n\n", array_map(
                fn ($item) => (string) ($item['content'] ?? ''),
                array_slice($matches, 0, 3)
            ));
        }

        $sources = collect($matches)
            ->unique('document_id')
            ->map(fn ($item) => [
                'title' => $item['title'] ?? 'DoR Official Document',
                'url' => $item['source_url'] ?? null,
            ])
            ->values()
            ->all();

        return response()->json([
            'ok' => true,
            'needs_consent' => false,
            'answer' => $answer,
            'source_label' => 'DoR Official Sources',
            'sources' => $sources,
            'provider_label' => $providerLabel,
        ]);
    }

    public function externalConsent(Request $request, string $subdomain)
    {
        $settings = ChatbotSetting::current();

        abort_unless($settings->enabled, 404);

        $request->validate([
            'question' => ['required', 'string', 'min:2', 'max:2000'],
        ]);

        abort_unless(
            (int) Office::getIdfromSubdomain($subdomain) > 0,
            404
        );

        // Explicit consent is received, but no external web-search
        // integration has been configured yet.
        return response()->json([
            'ok' => false,
            'code' => 'external_search_not_configured',
            'message' => 'बाह्य वेबसाइट खोज सेवा अहिले जडान गरिएको छैन।',
        ], 503);
    }
}

<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\ChatbotKnowledgeSource;
use App\Models\Office;
use App\Services\Chatbot\ChatbotKnowledgeIndexer;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Storage;
use Illuminate\Validation\Rule;
use Throwable;

class ChatbotKnowledgeController extends Controller
{
    private function ensureSuperAdmin(): void
    {
        abort_unless(
            Auth::check() && (int) Auth::user()->user_type_id === 2,
            403
        );
    }

    public function index()
    {
        $this->ensureSuperAdmin();

        $sources = ChatbotKnowledgeSource::where('source_type', '!=', 'appsheet')
            ->withCount('documents')
            ->orderByDesc('id')
            ->paginate(20);

        $offices = Office::getPublishedOffices();

        return view('chatbot_settings.admin.knowledge', compact(
            'sources',
            'offices'
        ));
    }

    public function upload(Request $request, ChatbotKnowledgeIndexer $indexer)
    {
        $this->ensureSuperAdmin();

        $uploadErrors = $_FILES['files']['error'] ?? [];
        if (!is_array($uploadErrors)) {
            $uploadErrors = [$uploadErrors];
        }

        foreach ($uploadErrors as $uploadError) {
            if (in_array((int) $uploadError, [
                UPLOAD_ERR_INI_SIZE,
                UPLOAD_ERR_FORM_SIZE,
            ], true)) {
                return back()->with('error', 'File is too large (maximum 300 MB).');
            }
        }

        $data = $request->validate([
            'office_id' => ['nullable', 'integer'],
            'files' => [
                'required',
                'array',
                'min:1',
                function ($attribute, $files, $fail) {
                    $totalBytes = 0;

                    foreach ((array) $files as $file) {
                        if ($file instanceof \Illuminate\Http\UploadedFile && $file->isValid()) {
                            $totalBytes += (int) $file->getSize();
                        }
                    }

                    if ($totalBytes > 300 * 1024 * 1024) {
                        $fail('Combined document size must not exceed 300 MB per upload.');
                    }
                },
            ],
            'files.*' => [
                'required',
                'file',
                'max:307200',
                'mimes:pdf,docx,xlsx,pptx',
            ],
            'is_public' => ['nullable', 'boolean'],
        ], [
            'office_id.integer' => 'Please select a valid office.',
            'files.required' => 'Please add at least one document.',
            'files.min' => 'Please add at least one document.',
            'files.*.file' => 'A selected file could not be uploaded.',
            'files.*.max' => 'A document is too large (maximum 300 MB).',
            'files.*.mimes' => 'Use PDF, DOCX, XLSX or PPTX documents.',
        ]);

        $officeId = isset($data['office_id']) && $data['office_id'] !== ''
            ? (int) $data['office_id']
            : null;

        if ($officeId !== null) {
            $officeExists = collect(Office::getPublishedOffices())
                ->contains(fn ($office) => (int) $office->id === $officeId);

            if (!$officeExists) {
                return back()->with('error', 'Please select a valid active office.');
            }
        }

        $indexed = 0;
        $failed = 0;

        foreach ($data['files'] as $file) {
            $source = null;
            $path = null;

            try {
                $source = ChatbotKnowledgeSource::create([
                    'office_id' => $officeId,
                    'source_type' => 'file',
                    'name' => $file->getClientOriginalName(),
                    'storage_disk' => 'local',
                    'storage_path' => null,
                    'language_code' => 'both',
                    'status' => 'pending',
                    'is_enabled' => true,
                    'is_public' => $request->boolean('is_public'),
                    'created_by' => Auth::id(),
                    'updated_by' => Auth::id(),
                ]);

                $folder = 'chatbot-knowledge/' . ($officeId ?? 'general');
                $path = $file->store($folder, 'local');

                if (!$path) {
                    $source->update([
                        'status' => 'failed',
                        'last_error' => 'Upload failed. Please try again.',
                    ]);
                    $failed++;
                    continue;
                }

                $source->update(['storage_path' => $path]);
            } catch (Throwable $e) {
                report($e);

                if ($source) {
                    $source->update([
                        'status' => 'failed',
                        'last_error' => 'Upload failed. Please try again.',
                    ]);
                }

                if ($path) {
                    Storage::disk('local')->delete($path);
                }

                $failed++;
                continue;
            }

            try {
                $result = $indexer->indexDocument([
                    'source_id' => $source->id,
                    'title' => $source->name,
                    'storage_disk' => 'local',
                    'storage_path' => $path,
                    'document_type' => 'document',
                    'mime_type' => $file->getMimeType(),
                    'source_key' => 'upload-' . $source->id,
                    'metadata' => [
                        'original_name' => $file->getClientOriginalName(),
                    ],
                ]);

                $success = in_array(
                    $result['status'] ?? '',
                    ['indexed', 'unchanged'],
                    true
                );

                $source->update([
                    'status' => $success ? 'indexed' : 'failed',
                    'last_error' => $success
                        ? null
                        : mb_substr(
                            (string) ($result['error'] ?? 'Indexing failed.'),
                            0,
                            500
                        ),
                    'updated_by' => Auth::id(),
                ]);

                $success ? $indexed++ : $failed++;
            } catch (Throwable $e) {
                report($e);

                $source->update([
                    'status' => 'failed',
                    'last_error' => 'Indexing failed. Please retry.',
                    'updated_by' => Auth::id(),
                ]);

                $failed++;
            }
        }

        return back();
    }

    public function registerUrl(Request $request)
    {
        $this->ensureSuperAdmin();

        $data = $request->validate([
            'office_id' => ['nullable', 'integer'],
            'is_public' => ['nullable', 'boolean'],
            'urls' => ['required', 'array', 'min:1'],
            'urls.*.type' => ['required', 'in:website,google_sheets'],
            'urls.*.name' => ['required', 'string', 'max:200'],
            'urls.*.source_url' => ['required', 'url', 'max:2000'],
        ], [
            'office_id.integer' => 'Please select a valid office.',
            'urls.required' => 'Please add at least one URL.',
            'urls.*.name.required' => 'Please enter a source name.',
            'urls.*.source_url.required' => 'Please enter a URL.',
            'urls.*.source_url.url' => 'Please enter a valid URL.',
        ]);

        $officeId = isset($data['office_id']) && $data['office_id'] !== ''
            ? (int) $data['office_id']
            : null;

        if ($officeId !== null) {
            $officeExists = collect(Office::getPublishedOffices())
                ->contains(fn ($office) => (int) $office->id === $officeId);

            if (!$officeExists) {
                return back()->withErrors([
                    'office_id' => 'Please select a valid active office.',
                ]);
            }
        }

        foreach ($data['urls'] as $index => $item) {
            $parts = parse_url($item['source_url']);

            if (
                !is_array($parts)
                || strtolower($parts['scheme'] ?? '') !== 'https'
                || empty($parts['host'])
                || isset($parts['user'])
                || isset($parts['pass'])
            ) {
                return back()->withErrors([
                    'urls' => 'Only HTTPS URLs without embedded credentials are accepted.',
                ]);
            }

        }

        foreach ($data['urls'] as $item) {

            ChatbotKnowledgeSource::create([
                'office_id' => $officeId,
                'source_type' => $item['type'],
                'name' => $item['name'],
                'source_url' => $item['source_url'],
                'language_code' => 'both',
                'status' => 'pending',
                'is_enabled' => true,
                'is_public' => $request->boolean('is_public'),
                'created_by' => Auth::id(),
                'updated_by' => Auth::id(),
            ]);
        }

        return back();
    }

    public function reindex(
        ChatbotKnowledgeSource $source,
        ChatbotKnowledgeIndexer $indexer
    ) {
        $this->ensureSuperAdmin();

        abort_unless($source->source_type === 'file', 422);

        if (!$source->is_enabled) {
            return back()->with('error', 'Enable this source before indexing.');
        }

        if (
            !$source->storage_path
            || !Storage::disk('local')->exists($source->storage_path)
        ) {
            return back()->with('error', 'The private source file is missing.');
        }

        try {
            $result = $indexer->indexDocument([
                'source_id' => $source->id,
                'title' => $source->name,
                'storage_disk' => 'local',
                'storage_path' => $source->storage_path,
                'document_type' => 'document',
                'source_key' => 'upload-' . $source->id,
            ]);

            $success = in_array($result['status'] ?? '', ['indexed', 'unchanged'], true);

            $source->update([
                'status' => $success ? 'indexed' : 'failed',
                'last_error' => $success
                    ? null
                    : mb_substr((string) ($result['error'] ?? 'Indexing failed.'), 0, 2000),
                'updated_by' => Auth::id(),
            ]);

            return back()->with(
                $success ? 'success' : 'error',
                $success ? 'Source indexed successfully.' : 'Indexing failed.'
            );
        } catch (Throwable $e) {
            report($e);

            $source->update([
                'status' => 'failed',
                'last_error' => 'Indexing failed. Check application logs.',
            ]);

            return back()->with('error', 'Indexing failed. Check application logs.');
        }
    }
