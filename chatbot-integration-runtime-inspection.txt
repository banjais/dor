===== PROVIDER REGISTRY =====
<?php

namespace App\Services\Chatbot;

use App\Models\ChatbotProvider;
use App\Models\ChatbotSetting;
use Illuminate\Support\Collection;


class ChatbotProviderRegistry
{
    /**
     * Return providers permitted by the global Chatbot settings.
     *
     * API keys are decrypted by the model cast and must never be
     * returned to controllers, views, or diagnostic output.
     */
    public function candidates(): Collection
    {
        $settings = ChatbotSetting::current();

        $externalAllowed = (bool) $settings->external_api_enabled;

        $providers = ChatbotProvider::query()
            ->where('is_enabled', true)
            ->orderBy('priority')
            ->get();

        $candidates = collect();

        foreach ($providers as $provider) {
            $type = strtolower((string) $provider->provider_type);

            if ($type === 'external' && ! $externalAllowed) {
                continue;
            }

            if (! in_array($type, ['external', 'internal'], true)) {
                continue;
            }

            if ($type === 'internal' && ! $settings->internal_ai_enabled) {
                continue;
            }

            $candidates->push([
                'id' => $provider->id,
                'code' => $provider->provider_code,
                'name' => $provider->provider_name,
                'type' => $type,
                'api_format' => $provider->api_format,
                'model' => $provider->model,
                'base_url' => $provider->base_url,
                'api_key' => $provider->api_key,
                'timeout_seconds' => $provider->timeout_seconds,
                'max_retries' => $provider->max_retries,
                'health_status' => $provider->health_status,
                'source' => 'provider_registry',
            ]);
        }

        /*
         * Backward compatibility:
         * Use the existing Primary / Secondary / Fallback settings
         * only when no enabled provider records are configured.
         */
        if ($candidates->isEmpty()) {
            foreach (['primary', 'secondary', 'fallback'] as $slot) {
                if (! $settings->{$slot . '_enabled'}) {
                    continue;
                }

                if (! $externalAllowed) {
                    continue;
                }

                $providerName = $settings->{$slot . '_provider'};

                if (! filled($providerName)) {
                    continue;
                }

                $candidates->push([
                    'id' => null,
                    'code' => $providerName,
                    'name' => ucfirst($providerName),
                    'type' => 'external',
                    'api_format' => 'openai_compatible',
                    'model' => $settings->{$slot . '_model'},
                    'base_url' => $settings->{$slot . '_base_url'},
                    'api_key' => $settings->{$slot . '_api_key'},
                    'timeout_seconds' => 30,
                    'max_retries' => 1,
                    'health_status' => 'unknown',
                    'source' => 'legacy_' . $slot,
                ]);
            }
        }

        return $candidates->values();
    }

    /**
     * External API requests are forbidden unless the Super Admin
     * has enabled the global switch.
     */
    public function externalRequestsAllowed(): bool
    {
        return (bool) ChatbotSetting::current()->external_api_enabled;
    }

    public function assertExternalRequestsAllowed(): void
    {
        if (! $this->externalRequestsAllowed()) {
            throw new ChatbotProviderException(
                'External AI requests are disabled by the global Chatbot setting.',
                null,
                'external_api_disabled'
            );
        }
    }
}
===== PROVIDER CLIENT =====
<?php

namespace App\Services\Chatbot;

use App\Models\ChatbotSetting;
use Illuminate\Http\Client\ConnectionException;
use Illuminate\Support\Facades\Http;

class ChatbotProviderClient
{
    public function __construct(
        private readonly ChatbotProviderRegistry $registry
    ) {}

    /**
     * Send a chat-completion request to an approved provider.
     *
     * This method deliberately does not retry. Routing and failover
     * will be handled by a separate service.
     */
    public function chat(array $provider, array $messages): array
    {
        $type = strtolower((string) ($provider['type'] ?? ''));

        if (! in_array($type, ['external', 'internal'], true)) {
            throw new ChatbotProviderException('Unsupported provider type.');
        }

        $settings = ChatbotSetting::current();

        // Final security check immediately before any HTTP request.
        if ($type === 'external') {
            $this->registry->assertExternalRequestsAllowed();
        } elseif (! $settings->internal_ai_enabled) {
            throw new ChatbotProviderException(
                'Internal AI is disabled by the Chatbot settings.',
                null,
                'internal_ai_disabled'
            );
        }

        if (($provider['api_format'] ?? '') !== 'openai_compatible') {
            throw new ChatbotProviderException(
                'Unsupported provider API format.'
            );
        }

        $baseUrl = trim((string) ($provider['base_url'] ?? ''));
        $model = trim((string) ($provider['model'] ?? ''));
        $apiKey = trim((string) ($provider['api_key'] ?? ''));

        if ($baseUrl === '' || $model === '') {
            throw new ChatbotProviderException(
                'Provider URL or model is not configured.'
            );
        }

        if ($type === 'external' && $apiKey === '') {
            throw new ChatbotProviderException(
                'External provider API key is not configured.'
            );
        }

        if (! is_array($messages) || $messages === []) {
            throw new ChatbotProviderException(
                'At least one chat message is required.'
            );
        }

        foreach ($messages as $message) {
            if (
                ! is_array($message) ||
                ! isset($message['role'], $message['content']) ||
                ! is_string($message['role']) ||
                ! is_string($message['content'])
            ) {
                throw new ChatbotProviderException(
                    'Invalid chat message structure.'
                );
            }
        }

        $parts = parse_url($baseUrl);

        if (
            $parts === false ||
            empty($parts['host']) ||
            isset($parts['user']) ||
            isset($parts['pass']) ||
            isset($parts['query']) ||
            isset($parts['fragment'])
        ) {
            throw new ChatbotProviderException(
                'Provider base URL is invalid.'
            );
        }

        $scheme = strtolower((string) ($parts['scheme'] ?? ''));
        $host = strtolower(trim((string) $parts['host'], '[]'));

        if (
            ($type === 'external' && $scheme !== 'https') ||
            ($type === 'internal' && ! in_array($scheme, ['http', 'https'], true))
        ) {
            throw new ChatbotProviderException(
                'Provider URL must use an approved HTTP scheme.'
            );
        }

        // External providers must not target local/private IP addresses.
        // Internal DoR AI may legitimately use private network addresses.
        if ($type === 'external') {
            $blockedLocalHosts = [
                'localhost',
                'localhost.localdomain',
            ];

            $isLocalHostname = in_array($host, $blockedLocalHosts, true)
                || str_ends_with($host, '.localhost')
                || str_ends_with($host, '.local')
                || str_ends_with($host, '.internal');

            $isPrivateOrReservedIp = filter_var(
                $host,
                FILTER_VALIDATE_IP,
                FILTER_FLAG_NO_PRIV_RANGE | FILTER_FLAG_NO_RES_RANGE
            ) === false && filter_var($host, FILTER_VALIDATE_IP) !== false;

            if ($isLocalHostname || $isPrivateOrReservedIp) {
                throw new ChatbotProviderException(
                    'External provider URL cannot target a local or private address.',
                    null,
                    'external_url_not_allowed'
                );
            }
        }

        $baseUrl = rtrim($baseUrl, '/');

        $endpoint = str_ends_with($baseUrl, '/chat/completions')
            ? $baseUrl
            : $baseUrl . '/chat/completions';

        $timeout = max(
            5,
            min(120, (int) ($provider['timeout_seconds'] ?? 30))
        );

        try {
            $request = Http::acceptJson()
                ->asJson()
                ->withOptions(['allow_redirects' => false])
                ->connectTimeout(min(10, $timeout))
                ->timeout($timeout);

            if ($apiKey !== '') {
                $request = $request->withToken($apiKey);
            }

            $response = $request->post($endpoint, [
                'model' => $model,
                'messages' => $messages,
            ]);
        } catch (ConnectionException) {
            throw new ChatbotProviderException(
                'Provider connection failed or timed out.',
                null,
                'provider_unavailable'
            );
        }

        if (! $response->successful()) {
            $status = $response->status();

            $code = match ($status) {
                401, 403 => 'authentication_error',
                408, 504 => 'timeout',
                429 => 'rate_limit',
                500, 502, 503 => 'provider_unavailable',
                default => 'provider_http_error',
            };

            // Do not expose the provider response body or API key.
            throw new ChatbotProviderException(
                'Provider request failed.',
                $status,
                $code
            );
        }

        $data = $response->json();

        $content = data_get($data, 'choices.0.message.content');

        if (! is_string($content) || trim($content) === '') {
            throw new ChatbotProviderException(
                'Provider returned an empty or unsupported response.'
            );
        }

        return [
            'content' => $content,
            'model' => data_get($data, 'model', $model),
            'usage' => data_get($data, 'usage', []),
            'provider_id' => $provider['id'] ?? null,
            'provider_code' => $provider['code'] ?? null,
            'provider_type' => $type,
        ];
    }
}
===== PROVIDER ROUTER =====
<?php

namespace App\Services\Chatbot;

use App\Models\ChatbotUsageLog;
use Illuminate\Support\Str;
use Throwable;

class ChatbotProviderRouter
{
    public function __construct(
        private readonly ChatbotProviderRegistry $registry,
        private readonly ChatbotProviderClient $client
    ) {}

    /**
     * Try configured providers in registry order.
     * A failed provider does not prevent trying the next provider.
     */
    public function chat(array $messages): array
    {
        $providers = $this->registry->candidates();

        if ($providers->isEmpty()) {
            throw new ChatbotProviderException(
                'No AI provider is currently enabled.',
                null,
                'no_provider_available'
            );
        }

        $requestId = (string) Str::uuid();

        $requestJson = json_encode(
            $messages,
            JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES
        );

        $requestHash = hash('sha256', $requestJson ?: '');

        foreach ($providers as $provider) {
            $maxRetries = max(
                0,
                min(3, (int) ($provider['max_retries'] ?? 0))
            );

            for ($attempt = 0; $attempt <= $maxRetries; $attempt++) {
                $startedAt = microtime(true);

                try {
                    $result = $this->client->chat($provider, $messages);

                    $latency = (int) round(
                        (microtime(true) - $startedAt) * 1000
                    );

                    $this->recordUsage(
                        $provider,
                        $requestId,
                        $requestHash,
                        $latency,
                        'success',
                        null,
                        null,
                        $result['usage'] ?? []
                    );

                    $result['request_id'] = $requestId;

                    return $result;
                } catch (ChatbotProviderException $e) {
                    $latency = (int) round(
                        (microtime(true) - $startedAt) * 1000
                    );

                    $this->recordUsage(
                        $provider,
                        $requestId,
                        $requestHash,
                        $latency,
                        'failed',
                        $e->errorCode,
                        $e->getMessage()
                    );

                    // Retry only temporary provider failures.
                    $retryable = in_array($e->errorCode, [
                        'rate_limit',
                        'timeout',
                        'provider_unavailable',
                    ], true);

                    if (! $retryable || $attempt >= $maxRetries) {
                        break;
                    }
                } catch (Throwable $e) {
                    $latency = (int) round(
                        (microtime(true) - $startedAt) * 1000
                    );

                    // Do not expose unexpected exception details.
                    $this->recordUsage(
                        $provider,
                        $requestId,
                        $requestHash,
                        $latency,
                        'failed',
                        'unexpected_provider_error',
                        'Provider request failed unexpectedly.'
                    );

                    break;
                }
            }
        }

        throw new ChatbotProviderException(
            'All configured AI providers failed or are unavailable.',
            null,
            'all_providers_failed'
        );
    }

    private function recordUsage(
        array $provider,
        string $requestId,
        string $requestHash,
        int $latency,
        string $status,
        ?string $errorCode,
        ?string $safeMessage,
        array $usage = []
    ): void {
        try {
            ChatbotUsageLog::create([
                'provider_id' => $provider['id'] ?? null,
                'request_id' => $requestId,
                'provider_slot' => substr(
                    (string) ($provider['source'] ?? 'registry'),
                    0,
                    20
                ),
                'model' => $provider['model'] ?? null,
                'status' => $status,
                'error_code' => $errorCode,
                'latency_ms' => $latency,
                'prompt_tokens' => data_get($usage, 'prompt_tokens'),
                'completion_tokens' => data_get($usage, 'completion_tokens'),
                'total_tokens' => data_get($usage, 'total_tokens'),
                'request_hash' => $requestHash,
                'safe_error_message' => $safeMessage,
                'created_at' => now(),
            ]);
        } catch (Throwable) {
            // Usage logging must not expose secrets or interrupt routing.
        }
    }
}
===== KNOWLEDGE INDEXER =====
<?php

namespace App\Services\Chatbot;

use App\Models\ChatbotKnowledgeDocument;
use App\Models\ChatbotKnowledgeSource;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;
use InvalidArgumentException;
use RuntimeException;

class ChatbotKnowledgeIndexer
{
    private const CHUNK_SIZE = 1200;
    private const CHUNK_OVERLAP = 150;

    public function __construct(
        private readonly ChatbotDocumentExtractor $extractor
    ) {
    }

    /**
     * Extract and index a local document.
     *
     * Required: title, storage_disk, storage_path.
     * Optional: source_id, office_id, language_id, document_type,
     * source_key, source_url, mime_type, source_updated_at, metadata.
     */
    public function indexDocument(array $data): array
    {
        foreach (['title', 'storage_disk', 'storage_path'] as $field) {
            if (!isset($data[$field]) || trim((string) $data[$field]) === '') {
                throw new InvalidArgumentException("Missing required field: {$field}");
            }
        }

        $disk = (string) $data['storage_disk'];
        $path = (string) $data['storage_path'];

        if (!in_array($disk, ['local', 'public'], true)) {
            throw new InvalidArgumentException('Unapproved storage disk.');
        }

        $extraction = $this->extractor->extract($disk, $path);

        // Never replace an existing successful index after extraction failure.
        if (($extraction['status'] ?? null) !== 'extracted'
            || trim((string) ($extraction['text'] ?? '')) === '') {
            return [
                'status' => 'failed',
                'error' => $extraction['error'] ?? 'No extractable text found.',
                'document' => null,
                'chunks' => 0,
            ];
        }

        $absolutePath = Storage::disk($disk)->path($path);

        if (!is_file($absolutePath) || !is_readable($absolutePath)) {
            return [
                'status' => 'failed',
                'error' => 'Document is not available for indexing.',
                'document' => null,
                'chunks' => 0,
            ];
        }

        $checksum = hash_file('sha256', $absolutePath);

        if ($checksum === false) {
            throw new RuntimeException('Unable to calculate document checksum.');
        }

        $source = null;

        if (!empty($data['source_id'])) {
            $source = ChatbotKnowledgeSource::findOrFail($data['source_id']);
        }

        $officeId = $data['office_id'] ?? $source?->office_id;
        $languageId = $data['language_id'] ?? null;
        $sourceId = $source?->id;

        $documentQuery = ChatbotKnowledgeDocument::query();

        if ($sourceId !== null) {
            $documentQuery->where('source_id', $sourceId);
        } else {
            $documentQuery->whereNull('source_id');
        }

        if (!empty($data['source_key'])) {
            $documentQuery->where('source_key', $data['source_key']);
        } else {
            $documentQuery
                ->where('storage_disk', $disk)
                ->where('storage_path', $path);
        }

        $existing = $documentQuery->first();

        // Identical content with existing chunks needs no re-indexing.
        if ($existing
            && $existing->checksum === $checksum
            && $existing->parser_status === 'indexed'
            && $existing->chunks()->exists()) {
            return [
                'status' => 'unchanged',
                'document' => $existing,
                'chunks' => $existing->chunks()->count(),
                'checksum' => $checksum,
            ];
        }

        $chunks = $this->splitText((string) $extraction['text']);

        if ($chunks === []) {
            return [
                'status' => 'failed',
                'error' => 'Document did not produce usable text chunks.',
                'document' => null,
                'chunks' => 0,
            ];
        }

        // All document changes and chunk replacement are atomic.
        $document = DB::transaction(function () use (
            $existing,
            $data,
            $source,
            $sourceId,
            $officeId,
            $languageId,
            $disk,
            $path,
            $checksum,
            $extraction,
            $chunks
        ) {
            $document = $existing
                ? ChatbotKnowledgeDocument::query()
                    ->whereKey($existing->id)
                    ->lockForUpdate()
                    ->firstOrFail()
                : new ChatbotKnowledgeDocument();

            $document->fill([
                'source_id' => $sourceId,
                'office_id' => $officeId,
                'language_id' => $languageId,
                'document_type' => $data['document_type'] ?? 'document',
                'source_key' => $data['source_key'] ?? null,
                'title' => $data['title'],
                'source_url' => $data['source_url'] ?? null,
                'storage_disk' => $disk,
                'storage_path' => $path,
                'mime_type' => $data['mime_type'] ?? null,
                'file_size' => filesize(Storage::disk($disk)->path($path)),
                'checksum' => $checksum,
                'source_updated_at' => $data['source_updated_at'] ?? null,
                'extracted_text' => $extraction['text'],
                'extraction_method' => $extraction['method'] ?? null,
                'parser_status' => 'indexed',
                'indexed_at' => now(),
                'last_error' => null,
                'metadata' => $data['metadata'] ?? null,
            ]);

            $document->save();

            // Old chunks are removed only inside the successful transaction.
            $document->chunks()->delete();

            foreach ($chunks as $number => $chunkText) {
                $document->chunks()->create([
                    'source_id' => $sourceId,
                    'office_id' => $officeId,
                    'language_id' => $languageId,
                    'chunk_number' => $number + 1,
                    'chunk_text' => $chunkText,
                    'token_count' => max(
                        1,
                        (int) ceil(mb_strlen($chunkText, 'UTF-8') / 4)
                    ),
                    'chunk_hash' => hash('sha256', $chunkText),
                    'status' => 'active',
                ]);
            }

            if ($source) {
                $source->forceFill([
                    'last_indexed_at' => now(),
                    'content_checksum' => $checksum,
                    'last_error' => null,
                ])->save();
            }

            return $document->fresh();
        });

        return [
            'status' => 'indexed',
            'document' => $document,
            'chunks' => count($chunks),
            'checksum' => $checksum,
            'method' => $extraction['method'] ?? null,
        ];
    }

    /**
     * Split Unicode text without cutting Nepali characters.
     */
    public function splitText(string $text): array
    {
        $text = trim(str_replace("\0", '', $text));

        if ($text === '') {
            return [];
        }

        $result = [];
        $length = mb_strlen($text, 'UTF-8');
        $start = 0;

        while ($start < $length) {
            $remaining = $length - $start;
            $take = min(self::CHUNK_SIZE, $remaining);
            $piece = mb_substr($text, $start, $take, 'UTF-8');

            if ($remaining > self::CHUNK_SIZE) {
                $newline = mb_strrpos($piece, "\n", 0, 'UTF-8');
                $space = mb_strrpos($piece, ' ', 0, 'UTF-8');
                $boundary = max(
                    $newline === false ? -1 : $newline,
                    $space === false ? -1 : $space
                );

                // Avoid producing very small chunks at an early boundary.
                if ($boundary >= (int) (self::CHUNK_SIZE * 0.60)) {
                    $take = $boundary + 1;
                    $piece = mb_substr($text, $start, $take, 'UTF-8');
                }
            }

            $piece = trim($piece);

            if ($piece !== '') {
                $result[] = $piece;
            }

            if ($start + $take >= $length) {
                break;
            }

            $nextStart = $start + $take - self::CHUNK_OVERLAP;

            // Guarantee forward progress.
            $start = max($start + 1, $nextStart);
        }

        return $result;
    }
}

<?php

namespace Tests\Feature;

use App\Models\ChatbotProvider;
use App\Models\ChatbotSetting;
use App\Models\ChatbotUsageLog;
use App\Services\Chatbot\ChatbotProviderClient;
use App\Services\Chatbot\ChatbotProviderException;
use App\Services\Chatbot\ChatbotProviderRouter;
use Illuminate\Database\Schema\Blueprint;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Schema;
use Tests\TestCase;

class ChatbotProviderTest extends TestCase
{
    protected function setUp(): void
    {
        parent::setUp();

        Schema::dropIfExists('chatbot_usage_logs');
        Schema::dropIfExists('chatbot_providers');
        Schema::dropIfExists('chatbot_settings');

        Schema::create('chatbot_settings', function (Blueprint $table) {
            $table->id();
            $table->string('scope', 30)->default('global')->unique();
            $table->boolean('enabled')->default(false);

    public function test_external_provider_returns_successful_response_when_enabled(): void
    {
        $settings = ChatbotSetting::current();
        $settings->external_api_enabled = true;
        $settings->save();

        Http::fake([
            'api.example.com/*' => Http::response([
                'model' => 'test-model',
                'choices' => [
                    ['message' => ['content' => 'DoR test answer']],
                ],
                'usage' => [
                    'prompt_tokens' => 10,
                    'completion_tokens' => 5,
                    'total_tokens' => 15,
                ],
            ], 200),
        ]);

        $result = app(ChatbotProviderClient::class)->chat([
            'type' => 'external',
            'api_format' => 'openai_compatible',
            'base_url' => 'https://api.example.com/v1',
            'model' => 'test-model',
            'api_key' => 'test-secret',
        ], $this->message());

        $this->assertSame('DoR test answer', $result['content']);
        $this->assertSame('test-model', $result['model']);
        $this->assertSame(15, $result['usage']['total_tokens']);

        Http::assertSent(function ($request) {
            return $request->url() === 'https://api.example.com/v1/chat/completions'
                && $request->hasHeader('Authorization', 'Bearer test-secret');
        });
    }

    public function test_router_fails_over_after_rate_limit_and_logs_without_api_key(): void
    {
        $settings = ChatbotSetting::current();
        $settings->external_api_enabled = true;
        $settings->save();

        $this->provider(
            'first-provider',
            1,
            'https://first.example.com/v1',
            'first-secret-key'
        );

        $this->provider(
            'second-provider',
            2,
            'https://second.example.com/v1',
            'second-secret-key'
        );

        Http::fake(function ($request) {
            if (str_contains($request->url(), 'first.example.com')) {
                return Http::response(['error' => 'rate limited'], 429);
            }

            return Http::response([
                'model' => 'second-provider-model',
                'choices' => [
                    ['message' => ['content' => 'Fallback answer']],
                ],
                'usage' => [
                    'prompt_tokens' => 12,
                    'completion_tokens' => 8,
                    'total_tokens' => 20,
                ],
            ], 200);
        });

        $result = app(ChatbotProviderRouter::class)->chat($this->message());

        $this->assertSame('Fallback answer', $result['content']);
        $this->assertSame('second-provider', $result['provider_code']);

        $this->assertSame(2, ChatbotUsageLog::count());
        $this->assertSame(
            ['failed', 'success'],
            ChatbotUsageLog::orderBy('id')->pluck('status')->all()
        );
        $this->assertSame(
            'rate_limit',
            ChatbotUsageLog::where('status', 'failed')->value('error_code')
        );

        $logs = json_encode(ChatbotUsageLog::all()->toArray());

        $this->assertStringNotContainsString('first-secret-key', $logs);
        $this->assertStringNotContainsString('second-secret-key', $logs);
    }
